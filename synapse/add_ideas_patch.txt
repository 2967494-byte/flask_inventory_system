## 3. Обновить функцию fetchAll
Заменить блок fetchAll на:
const fetchAll = async () => {
  if (!token) return
  setLoading(true)
  setAuthError(false)
  try {
    const config = { headers: { Authorization: `Bearer ${token}` } }
    const [pRes, tRes, trRes, sRes, nRes] = await Promise.all([
      axios.get(`${BASE_URL}/projects`, config),
      axios.get(`${BASE_URL}/tasks`, config),
      axios.get(`${BASE_URL}/transactions`, config),
      axios.get(`${BASE_URL}/dashboard/stats`, config),
      axios.get(`${BASE_URL}/notes`, config)  // Добавили загрузку идей
    ])
    setProjects(pRes.data)
    setTasks(tRes.data)
    setTransactions(trRes.data)
    setStats(sRes.data)
    setNotes(nRes.data)  // Устанавливаем идеи
  } catch (err: any) {
    console.error("Ошибка синхронизации:", err)
    if (err.response?.status === 401) {
      setAuthError(true)
      setToken(null)
      localStorage.removeItem('synapse_token')
    }
  }
  setLoading(false)
}
